def _status_error() -> httpx.HTTPStatusError:
    request = httpx.Request("GET", "http://localhost/v1.41/info")
    return httpx.HTTPStatusError(
        "Server error '500'", request=request, response=httpx.Response(500, request=request)
    )


async def test_http_status_error_ends_the_epoch(
    prepared, controller_settings, engine_boundary, monkeypatch
):
    """ADR C2 names 'status' explicitly; the repo matrix parametrizes only connect and timeout."""
    controller, record = prepared
    with monkeypatch.context() as patch:
        patch.setattr(engine_boundary, "volume_path", raising(_status_error()))
        with pytest.raises(httpx.HTTPStatusError):
            await controller.validate(1002, record)
    assert generation_on_disk(controller_settings) == record.fence.generation + 1
    with pytest.raises(ValueError, match="generation changed"):
        await controller.validate(1002, record)


async def test_describe_and_authenticate_still_answer_after_a_failed_persist(
    prepared, controller_settings, engine_boundary, monkeypatch
):
    """The operator contract says a controller that cannot persist 'refuses every request'."""
    controller, record = prepared
    with monkeypatch.context() as patch:
        patch.setattr(
            ManagedComposeController,
            "_persist",
            lambda self, state: (_ for _ in ()).throw(OSError("state device is read-only")),
        )
        patch.setattr(engine_boundary, "volume_path", raising(_status_error()))
        with pytest.raises(OSError, match="read-only"):
            await controller.validate(1002, record)
    grant = controller.authenticate(1002)
    root = controller.describe(1002)
    print(
        f"\nafter a failed persist: authenticate -> {grant.operator}; "
        f"describe -> root_id {root.root_id}; disk generation "
        f"{generation_on_disk(controller_settings)} (in-memory advanced)"
    )
