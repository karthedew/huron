async def test_a1_establish_then_old_record_is_refused(prepared, controller_settings):
    """A2/A1: minting a fresh epoch makes the previously admitted record inadmissible."""
    controller, record, _, _ = prepared
    await controller.establish(1002, UUID(int=3))
    with pytest.raises(ValueError, match="generation changed"):
        await controller.validate(1002, record)


async def test_a2_forged_record_naming_the_live_generation_but_old_report(
    prepared, controller_settings, engine_boundary
):
    """A record whose fence is edited to the live generation but whose report is not is refused."""
    controller, record, inventory, store = prepared
    fresh = await controller.establish(1002, UUID(int=3))
    forged = record.model_copy(update={"fence": fresh})
    path = engine_boundary.volume / "protocol/000001.json"
    path.chmod(0o600)
    path.write_bytes(forged.canonical)
    with pytest.raises(ValueError, match="disagrees with its inventory"):
        await controller.validate(1002, forged)


async def test_a3_abandon_with_stale_evidence_then_validate_still_admits(
    prepared, controller_settings
):
    """A stale receipt passed to abandon must not end the live epoch."""
    controller, record, _, _ = prepared
    stale = record.fence.model_copy(update={"generation": record.fence.generation - 1})
    controller.abandon(1002, stale)
    assert generation_on_disk(controller_settings) == record.fence.generation
    await controller.validate(1002, record)


async def test_a4_foreign_operator_receipt_cannot_end_the_epoch(prepared, controller_settings):
    """A receipt naming another operator is not the live receipt, so abandon is a no-op."""
    controller, record, _, _ = prepared
    foreign = record.fence.model_copy(update={"operator": "local:9999"})
    controller.abandon(1002, foreign)
    assert generation_on_disk(controller_settings) == record.fence.generation
    await controller.validate(1002, record)


async def test_a5_check_after_abandon_is_refused(prepared, controller_settings):
    """After abandon nothing observes the old epoch as live."""
    controller, record, _, _ = prepared
    controller.abandon(1002, record.fence)
    with pytest.raises(ValueError, match="generation changed"):
        controller.check(1002, record.fence)
    with pytest.raises(ValueError, match="generation changed"):
        await controller.validate(1002, record)


async def test_a6_restart_between_scan_and_validate_refuses_the_record(
    composed, controller_settings, engine_boundary
):
    """A controller restart after a complete scan refuses the record and forces a rescan."""
    controller, service = composed
    record = await service.prepare(UUID(int=4))
    before = generation_on_disk(controller_settings)
    restarted = ManagedComposeController(controller_settings)
    assert generation_on_disk(controller_settings) == before + 1
    with pytest.raises(ValueError, match="generation changed"):
        await restarted.validate(1002, record)
    # The pre-restart object is not a live controller in production: serve() holds an
    # exclusive flock on the state lock, so only one process ever owns this state.
    print("\nA6 pre-restart in-process object still admits:", end=" ")
    try:
        await controller.validate(1002, record)
        print("yes (single-owner flock is the only thing preventing two live controllers)")
    except ValueError as error:
        print(f"no ({error})")


async def test_a7_second_coordinator_is_refused_without_ending_the_first_epoch(
    composed, controller_settings, engine_boundary, monkeypatch
):
    """A contending coordinator is refused by the kernel claim, not by burning the epoch."""
    controller, service = composed
    other = MaintenanceStore(engine_boundary.volume, controller_settings.owner_uid)
    started = asyncio.Event()
    released = asyncio.Event()
    outcome: list[object] = []

    real_next = CutoverService._next_sequence

    def slow_next_sequence(self):
        started.set()
        # busy-wait so the contending task can run
        return real_next(self)

    monkeypatch.setattr(CutoverService, "_next_sequence", slow_next_sequence)

    async def contender():
        await started.wait()
        try:
            with other.claim():
                outcome.append("claimed")
        except BlockingIOError as error:
            outcome.append(error)
        released.set()

    task = asyncio.create_task(contender())
    record = await service.prepare(UUID(int=4))
    await task
    assert outcome and isinstance(outcome[0], BlockingIOError), outcome
    await service.validate(record)


async def test_a8_persist_failure_during_observation_still_refuses_the_receipt(
    prepared, controller_settings, engine_boundary, monkeypatch
):
    """If the epoch-end write itself fails, the receipt must still never be admitted again."""
    controller, record, _, _ = prepared
    disk_before = generation_on_disk(controller_settings)
    with monkeypatch.context() as patch:
        patch.setattr(
            ManagedComposeController,
            "_persist",
            lambda self, state: (_ for _ in ()).throw(OSError("state device is read-only")),
        )
        patch.setattr(engine_boundary, "containers", (writer(engine_boundary),))
        with pytest.raises(OSError, match="read-only"):
            await controller.validate(1002, record)
    disk_after = generation_on_disk(controller_settings)
    print(f"\nA8 disk generation before={disk_before} after={disk_after} (unchanged on disk)")
    with pytest.raises(ValueError, match="generation changed"):
        await controller.validate(1002, record)
    restarted = ManagedComposeController(controller_settings)
    print(f"A8 after restart, disk generation={generation_on_disk(controller_settings)}")
    with pytest.raises(ValueError, match="generation changed"):
        await restarted.validate(1002, record)


async def test_a9_report_swap_under_the_live_epoch_is_refused(
    prepared, controller_settings, engine_boundary

    """Replacing the bound report with another valid report is refused by the digest binding."""

    other = ProtocolInventory(

        run_id=UUID(int=7),
        fence=record.fence,
        revisions=(),
        generations=(),

    path = engine_boundary.volume / "protocol" / record.report_path
    path.chmod(0o600)
    path.write_bytes(other.canonical)
    with pytest.raises(ValueError, match="disagrees with its inventory"):
        await controller.validate(1002, record)
